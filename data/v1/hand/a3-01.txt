@create style=detailed domain=a3_carrom
Scoreboard for carrom between me and my brother. Tap to add coins for each of us, the queen is 3 points, and keep count of boards won.
---
app "Carrom" board plum
table pts player pick Me|Bro, coins num 1..9 = 1
table boardwin winner pick Me|Bro
let me_s = sum(pts.coins where player = Me)
let bro_s = sum(pts.coins where player = Bro)
row gap 2
  stack grow align center card
    "Me" bold
    text me_s big
    button "+1 coin" tap add pts player = Me, coins = 1
    button "Queen +3" tap add pts player = Me, coins = 3
  stack grow align center card
    "Bro" bold
    text bro_s big
    button "+1 coin" tap add pts player = Bro, coins = 1
    button "Queen +3" tap add pts player = Bro, coins = 3
row gap 2
  "Boards won: Me" count(boardwin where winner = Me)
  "Bro" count(boardwin where winner = Bro)
button "I won this board" tap add boardwin winner = Me then remove all pts
button "Bro won this board" tap add boardwin winner = Bro then remove all pts
===

@create style=custom_ui domain=a3_badminton
badminton match to 21 points, two sides, big tap areas that add a point, show who is leading and sets won
---
app "Badminton" ball ocean
table point side pick Left|Right
table setwon side pick Left|Right
let l = count(point where side = Left)
let r = count(point where side = Right)
let over = (l >= 21 or r >= 21) and abs(l - r) >= 2
grid 2 gap 2
  stack card align center pad 3 fill soft tap add point side = Left
    "Left" faint
    text l big
  stack card align center pad 3 fill soft tap add point side = Right
    "Right" faint
    text r big
if over
  if(l > r, "Left", "Right") "wins the set" head color good
  button "Next set" tap add setwon side = if(l > r, Left, Right) then remove all point
else
  if(l = r, "Level", if(l > r, "Left leads", "Right leads")) faint
"Sets: Left" count(setwon where side = Left) "| Right" count(setwon where side = Right) bold
button "Undo point" tap remove last(point)
===
