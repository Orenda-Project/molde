@create style=custom_ui domain=a3_interval
interval trainer: 40 seconds work, 20 seconds rest, counts rounds, big timer that changes colour between work and rest
---
app "Intervals" bolt ink
setting work dur = 40s "Work"
setting rest dur = 20s "Rest"
state left dur = 40s
state going bool = no
state working bool = yes
state rounds num = 0
every 1s if going set left = left - 1s
when going and left <= 0s set rounds = rounds + if(working, 0, 1) then set left = if(working, rest, work) then set working = not working
stack align center card pad 3 fill if(working, accent, good)
  if(working, "WORK", "REST") head
  text left big
  "Round" rounds + 1 faint
row gap 1
  button if(going, "Pause", "Start") tap set going = not going
  button "Reset" tap set going = no then set left = work then set working = yes then set rounds = 0
===

---
~ setting work dur = 40s => setting work dur = 30s
~ setting rest dur = 20s => setting rest dur = 15s
~ state left dur = 40s => state left dur = 30s
