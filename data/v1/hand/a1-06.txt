@create style=multi_feature domain=a1_umrah
umrah preparation with two pages: a packing checklist by type (clothes, documents, medicine, other) and a savings page with a thermometer for the trip money
---
app "Umrah" mosque ocean
table pack item text, kind pick Clothes|Documents|Medicine|Other = Clothes, done bool
table saved amount money
setting cost money = 400000 "Trip cost"
screen packs "Pack" box
  "Packed" count(pack where done = yes) "of" count(pack) head
  each options pack.kind as k
    each pack where kind = k as p
      row tap edit p swipe remove p
        text p.item grow
        toggle p.done
  form pack, "Add item"
screen money "Savings" coin
  let have = sum(saved.amount)
  let p = clamp(have / cost, 0, 1)
  row gap 3
    draw 30, 100 width 40
      rect 8, 4, 14, 92 stroke ink thick 2 radius 7
      rect 8, 96 - 92 * p, 14, 92 * p fill good radius 7
    stack grow
      text have big
      "of" cost faint
  quick saved, amount, [5000, 10000, 20000]
  form saved
===

@create style=custom_ui domain=a1_macro_pie
what is on my plate: log meals with protein, carbs and fat in grams and draw a ring split into three coloured parts for today
---
app "Plate" food mint
table meal item text, protein num 0..300 = 0, carbs num 0..500 = 0, fat num 0..300 = 0
let p = sum(meal.protein today)
let c = sum(meal.carbs today)
let f = sum(meal.fat today)
let t = max(p + c + f, 1)
draw 100, 100 width 200
  circle 50, 50, 40 stroke soft thick 14
  arc 50, 50, 40, 0, 360 * p / t stroke accent thick 14
  arc 50, 50, 40, 360 * p / t, 360 * (p + c) / t stroke good thick 14
  arc 50, 50, 40, 360 * (p + c) / t, 360 stroke warn thick 14
  label 50, 50, p + c + f "g" head
row gap 2
  "Protein" p color accent
  "Carbs" c color good
  "Fat" f color warn
form meal
