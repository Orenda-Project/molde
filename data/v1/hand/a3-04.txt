@create style=custom_ui domain=a3_thermo_save
savings thermometer for my umrah fund that goes red to the top as I put money in, with a goal I can change
---
app "Umrah fund" coin mint
table dep amount money
setting goal money = 500000 "Goal"
let have = sum(dep.amount all)
let p = clamp(have / goal, 0, 1)
row gap 3
  draw 40, 120 width 56
    rect 14, 6, 12, 90 stroke ink thick 2 radius 6
    circle 20, 104, 12 fill red
    rect 16, 100 - 92 * p, 8, 4 + 92 * p fill red
  stack grow gap 1
    text have big
    "of" goal faint tap edit goal
    round(p * 100) "% saved" bold
quick dep, amount, [5000, 10000, 25000]
form dep
log dep
