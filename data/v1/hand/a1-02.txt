@create style=custom_ui domain=a1_meal_plan
a weekly meal plan, one card per weekday with breakfast lunch and dinner, today's card highlighted
---
app "Meal Plan" food mint
table meal day pick Mon|Tue|Wed|Thu|Fri|Sat|Sun, slot pick Breakfast|Lunch|Dinner, dish text
each options meal.day as d
  stack card gap 0 fill if(d = dayname(today), soft, surface)
    text d bold
    each meal where day = d sort slot as m
      row tap edit m swipe remove m
        text m.slot faint
        text m.dish grow
form meal

@create style=multi_feature domain=a1_pantry
pantry stock: item, how many, minimum I want to keep, plus and minus buttons, search box, and a count of what is running low
---
app "Pantry" box slate
table stock name text, qty num 0..1000, low num 0..1000 = 2, unit text = "pcs"
state q text = ""
let lowc = countof(stock, it.qty <= it.low)
"Running low" lowc big color if(lowc > 0, warn, good)
field q "Search"
each stock where has(name, q) sort name as s
  row card gap 1 tap edit s
    stack grow
      text s.name bold
      s.qty s.unit color if(s.qty <= s.low, bad, muted)
    button "-" tap set s.qty = s.qty - 1
    button "+" tap set s.qty = s.qty + 1
form stock, "Add item"
===
