@create style=multi_feature domain=a1_bp_log
blood pressure log: top and bottom number, pulse, tell me if the last reading is high, and a chart of the last 2 weeks
---
app "BP Log" heart ocean
table bp sys num 60..250 "Top number", dia num 30..150 "Bottom number", pulse num? 30..220
let high = last(bp.sys) >= 140 or last(bp.dia) >= 90
text last(bp.sys) "/" last(bp.dia) big color if(high, warn, good)
if high
  "Last reading is high. Rest for a few minutes and measure again." faint
"Average this week" round(avg(bp.sys last 7 days)) "/" round(avg(bp.dia last 7 days)) faint
trend bp last 14 days sort at, it.sys
form bp, "Save reading"
log bp
===
