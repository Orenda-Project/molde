---

state bill num = 0

@create style=multi_feature domain=expense_split
split expenses with friends on a trip: who paid, how much, and show what each person owes in the end
---
app "Trip split" users ocean
table friend name text
table pay who ref friend, amount money, what text?
let total = sum(pay.amount all)
let share = total / max(count(friend), 1)
"Total spent" total big
"Each share" share faint
each friend as f
  row card
    f.name bold grow
    "paid" sum(f.pay.amount) faint
    if(sum(f.pay.amount) >= share, "gets back", "owes") abs(sum(f.pay.amount) - share) bold
form friend, "Add friend"
form pay, "Add payment"
log pay
===
