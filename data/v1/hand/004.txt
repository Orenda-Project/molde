@create style=detailed domain=n_compound
Compound growth calculator: I type my starting money, the yearly profit percent and the number of years, and it shows what I will have at the end.
---
app "Growth" coin mint
state start num = 100000
state rate num = 10
state years num = 5
field start "Starting money"
field rate "Yearly profit %"
slider years 1..30
let end = start * pow(1 + rate / 100, years)
"After" years "years" faint
"You will have" round(end) big
"Profit" round(end - start) bold
===
@create style=custom_ui domain=n_flash_shuffle
flashcards in random order, show the front, tap to see the answer, a button for the next card
---
app "Quiz cards" brain plum
table card front text, back text
state shown bool
state pos num = 1
each card shuffle top 1 empty "Add some cards below"
  stack card gap 2 tap set shown = not shown

@create style=multi_feature domain=n_weight_plot
weight tracker with a line chart that has axes and gridlines, my target as a setting, and the change since the first weigh-in
---
app "Weight" scale mint
table weigh kg num 20..300
setting target num = 70 "Target (kg)"
let first_kg = first(weigh.kg)
let now_kg = last(weigh.kg)
text now_kg "kg" big
"Change since start" fixed(now_kg - first_kg, 1) "kg" faint
plot 0..max(count(weigh), 2), 40..120 axes gridlines
  poly stroke accent thick 2
    each weigh sort at
      pt index, it.kg
form weigh
===

@create style=custom_ui domain=n_score_join
cricket scorer that shows the score like 120/3 and the overs, buttons for runs and a wicket
---
app "Scorer" ball ocean
table ball runs num 0..6, out bool
let r = sum(ball.runs all)
let w = count(ball where out = yes)
join(r, "/", w) big
"Overs" floor(count(ball) / 6) "." count(ball) % 6 faint
row gap 1
  each [0, 1, 2, 3, 4, 6]
    button it tap add ball runs = it
button "Wicket" fill bad tap add ball runs = 0, out = yes
button "Undo" tap remove last(ball)
===
