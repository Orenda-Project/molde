@create style=multi_feature domain=a3_trip_planner
trip planner with 3 tabs: day by day plan, trip expenses with a total, and a packing checklist
---
app "Trip" plane ocean
table plan day date, what text
table spent label text, amount money
table pack item text, packed bool
screen days "Plan" calendar
  each plan sort day as p
    row card gap 2 tap edit p swipe remove p
      p.day bold
      p.what grow
  form plan, "Add to plan"
screen money "Money" wallet
  "Spent so far" sum(spent.amount all) big
  form spent
  list spent
screen bag "Pack" box
  "Packed" count(pack where packed = yes) "of" count(pack) head
  each pack sort at as i
    row card
      i.item grow
      toggle i.packed
  form pack, "Add item"
