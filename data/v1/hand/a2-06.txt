@create style=detailed domain=a2_compound_savings
I want to see what happens if I save a fixed amount every month for some years at a yearly return. Let me change the amount, the percent and the years, and show the final amount with a bar for each year.
---
app "Growth" chart mint
state pay money = 5000
state rate num = 10
state years num = 10
field pay "Saving each month"
"Yearly return %" rate faint
slider rate 0..30
"Years" years faint
slider years 1..30
let fv(y num) = if(rate = 0, pay * 12 * y, pay * (pow(1 + rate / 1200, 12 * y) - 1) / (rate / 1200))
"After" years "years" faint
compact(fv(years)) big
"You put in" compact(pay * 12 * years) faint
plot 0..years, 0..fv(years) axes
  each 1..years
    rect index + 0.15, 0, 0.7, fv(it) fill accent radius 2
    tick index + 0.5, it

@create style=multi_feature domain=a2_emi_calc
loan EMI calculator with the real compound interest formula: loan amount, yearly interest percent, months with a slider, shows the monthly instalment, total paid and total interest
---
app "EMI" calc ocean
state loan money = 0
state rate num = 12
state months num = 24
field loan "Loan amount"
field rate "Yearly interest %"
"Months" months faint
slider months 6..120
let r = rate / 1200
let emi = if(r = 0, loan / max(months, 1), loan * r * pow(1 + r, months) / (pow(1 + r, months) - 1))
"Monthly instalment" round(emi) big
"Total paid" grouped(round(emi * months)) faint
"Interest" grouped(round(emi * months - loan)) faint

@create style=multi_feature domain=a2_class_quiz

---
app "Class quiz" school plum
table q question text, a text, b text, right pick A|B = A
state i num = 1
state score num = 0
if i > count(q)
  "Finished" head
  "Score" score "of" count(q) big
  button "Start again" tap set i = 1 then set score = 0
else
  "Question" i "of" count(q) faint
  let cq = nth(q sort at, i)
  cq.question big
  button cq.a tap set score = score + if(cq.right = A, 1, 0) then set i = i + 1
  button cq.b tap set score = score + if(cq.right = B, 1, 0) then set i = i + 1
form q, "Add question"
