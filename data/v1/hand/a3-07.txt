@create style=multi_feature domain=a3_chores_tabs
household chores in three tabs: Today shows chores due today with a tick, All lists every chore with how often, Add has the form
---
app "House chores" home mint
table chore name text, every num 1..30 = 7
table done chore ref chore
screen td "Today" check
  each chore sort at as c
    if since(c.done) >= c.every or count(c.done) = 0
      row card gap 2
        c.name bold grow
        toggle c.done
screen all "All" list
  each chore sort name as c
    row card gap 2 tap edit c swipe remove c
      c.name grow
      "every" c.every "days" faint
      "last" since(c.done) "d ago" faint
screen add "Add" plus
  form chore, "Save chore"
===

@create style=custom_ui domain=a3_sunday_chore_wheel
chore wheel for the family, names on a table, a spinner arrow that points at a random family member after I tap, with the name shown below
---

table fam name text
state ang num = 0
state pk num = 1
let n = max(count(fam), 1)
draw 100, 100 width 220 tap set pk = random(n) then set ang = (pk - 1) * 360 / n + 180 / n
  circle 50, 50, 44 stroke soft thick 4
  each 1..n
