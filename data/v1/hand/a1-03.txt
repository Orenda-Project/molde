@create style=multi_feature domain=a1_plant_water
my plants: name, water every how many days, a button when I water, tell me which ones need water today
---
app "Plant Care" plant mint
table plant name text, every num 1..60 = 3 unit "days"
table watered plant ref plant
let thirsty = countof(plant, since(it.watered) >= it.every or count(it.watered) = 0)
"Need water" thirsty big color if(thirsty > 0, warn, good)
each plant sort name as p
  row card gap 1 tap edit p
    stack grow
      text p.name bold
      "every" p.every "days" faint
    if since(p.watered) >= p.every or count(p.watered) = 0
      "Thirsty" color warn
    button "Watered" tap add watered plant = p
form plant, "Add plant"
===
