@create class=sensible_calc style=calc
loan monthly payment calculator
plan: # need: field amount, field rate, field months
---
app "Loan payment" calc blue
state amount money = 10000
state rate num = 10
state months num = 24
field amount "Loan amount"
field rate "Yearly interest (%)"
field months "Months"
let r = rate / 1200
let pay = if(r = 0, amount / months, amount * r / (1 - pow(1 + r, 0 - months)))
"Monthly payment" round(pay * 100) / 100 big
"Total paid" round(pay * months * 100) / 100 head
"Interest paid" round((pay * months - amount) * 100) / 100 head
===

savings growth with monthly deposit and yearly interest over 5 years
plan: # need: field deposit, field rate
---
app "Savings growth" coin teal
state dep money = 200
state rate num = 6
field dep "Monthly deposit"
field rate "Yearly interest (%)"
let m = rate / 1200
let n = 60
let fv = if(m = 0, dep * n, dep * (pow(1 + m, n) - 1) / m)
"After 5 years" round(fv) big
"You put in" dep * n head
