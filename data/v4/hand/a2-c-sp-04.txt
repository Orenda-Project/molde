@create class=rich_app style=rich
sleep tracker
plan: # need: field bed, field wake, goal, chart, average, list
---
app "Sleep tracker" bed plum dark
table night bed time, wake time, quality num 1..5 = 3
setting target num 4..12 = 8 "Sleep goal (h)"
let hrs(n night) = (n.wake - n.bed) / 1h
topbar night, "Log night"
row gap 2
  statcard round(sum(night last 7 days, hrs(it)) * 10 / max(count(night last 7 days), 1)) / 10, "h average this week", target
  statcard round(avg(night.quality last 7 days) * 10) / 10, "average quality"
section "Hours per night"
  bars night last 7 days sort at, hrs(it), it.at
each night top 7 as n empty "No nights logged yet."
  row card gap 2 tap edit n swipe remove n
    n.bed "to" n.wake grow
    fixed(hrs(n), 1) "h" bold color if(hrs(n) >= target, good, warn)
