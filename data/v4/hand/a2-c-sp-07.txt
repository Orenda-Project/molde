@create class=rich_app style=rich
a savings tracker
plan: # need: field amount, setting target, goal, chart, list
---
app "Savings" coin green
table save amount money, note text?
setting target money = 5000 "Savings target"
topbar save, "Add savings"
row gap 2
  statcard sum(save.amount), "saved", target
  statcard sum(save.amount month), "saved this month"
goal sum(save.amount), target, "of target"
"Still to go" max(target - sum(save.amount), 0) head
section "Growth", "total saved over time"
  trend save last 12 months sort at, it.amount
list save by day, "Nothing saved yet. Tap Add savings."
===

@create class=rich_app style=rich
a daily planner
plan: # need: field task, field time, tick, total today, list
---
app "Daily planner" calendar mint
table task what text, at_time time = 9:00, done bool
state d date = today
topbar task, "Add task"
row align center gap 1
  button "<" tap set d = d - 1
  d bold grow align center
  button ">" tap set d = d + 1
statcard count(task on d where done = yes), "done of" count(task on d), count(task on d)
each task on d sort at_time as t empty "Nothing planned for this day."
  row card gap 2 tap edit t swipe remove t
    toggle t.done
    t.at_time faint
    t.what bold grow

@create class=rich_app style=rich
a restaurant tab
plan: # need: field item, field price, field qty, total, tick, list
---
app "Restaurant tab" cup red
table order item text, price money, qty num 1..50 = 1, served bool
let total = sum(order, it.price * it.qty)
topbar order, "Add order"
row gap 2
  statcard total, "bill so far"
  statcard count(order where served = no), "still to serve"
"With 10% service" round(total * 110) / 100 head
each order sort served as o empty "No orders yet."
  row card gap 2 tap edit o swipe remove o
    toggle o.served
    stack grow gap 0
      o.item bold
      o.qty "x" o.price faint
    o.price * o.qty bold
===
