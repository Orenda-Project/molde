@base id=bakery class=long_base
request: a bakery sales app: my items with prices and kind, record sales, daily takings against a target, best sellers and a weekly chart
---
app "Bakery" bread pink
table item name text, price money, kind pick Bread|Cake|Cookies = Bread
table sale item ref item, qty num 1..100 = 1, paid bool
setting target money = 20000 "Daily target"
let takings = sum(sale today, it.qty * it.item.price)
let unpaid = count(sale where paid = no)
topbar sale, "New sale"
statcard takings, "taken today", target
if takings >= target
  "Target reached. Good day at the oven." head color good
section "Today", "sales"
  each sale today as s empty "No sales yet today."
    row card gap 2 tap edit s swipe remove s
      stack grow gap 0
        s.item.name bold
        s.qty "x" s.item.price faint
      toggle s.paid
  "Not paid yet:" unpaid faint
section "Takings this week"
  daybars last 7 days, sum(sale on it, it.qty * it.item.price)
section "Best sellers", "this month"
  each item sort sum(it.sale.qty month) desc top 5 as i
    row gap 2
      i.name grow
      sum(i.sale.qty month) bold
section "By kind"
  each options item.kind as k
    row gap 1
      k grow
      count(sale month where item.kind = k) faint
section "Items"
  each item sort name as i
    row card gap 2 tap edit i swipe remove i
      i.name bold grow
      i.price faint
      i.kind faint
form item, "Add item"
