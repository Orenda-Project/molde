@create class=keep_specifics style=specifics

plan: # need: field opening, field sales, field expenses, list
---
app "Stall float" cart green
table day_row opening money, sales money, expenses money
let closing(d day_row) = d.opening + d.sales - d.expenses
let profit(d day_row) = d.sales - d.expenses
each day_row top 14 as d empty "No days yet."
  stack card gap 1 tap edit d swipe remove d
    d.at faint
    row
      "Closing cash" grow
      closing(d) bold
    row
      "Profit" grow
      profit(d) bold color if(profit(d) >= 0, good, bad)
form day_row, "Add day"
