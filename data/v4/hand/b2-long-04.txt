@base id=committee class=long_base
request: village savings committee: members, monthly contribution of each, whose turn it is to receive the pot, and who is late
---
app "Committee" users orange
setting share money = 5000 "Monthly share"
table member name text, turn num 1..60, phone text?
table deposit member ref member, amount money
let pot = count(member) * share
topbar deposit, "Record deposit"
statcard sum(deposit.amount month), "collected this month", pot
"This month's pot" pot big
each member sort turn as m
  if m.turn = (month(today) % count(member)) + 1
    row card gap 2 fill soft
      icon trophy
      m.name bold grow
      "gets the pot this month" faint
section "Paid", "this month"
  each member where sum(deposit.amount month) >= share sort name as m empty "Nobody has paid yet."
    row gap 2
      icon check
      m.name grow
      sum(m.deposit.amount month) faint
section "Late"
  each member where sum(deposit.amount month) < share sort name as m
    row card gap 2 tap edit m
      m.name bold grow
      share - sum(m.deposit.amount month) bold color bad
      button "Got it" tap add deposit member = m, amount = share - sum(m.deposit.amount month)
section "Order of turns"
  each member sort turn as m
    row gap 2 swipe remove m
      m.turn faint
      m.name grow
      m.phone faint
section "Deposits by month"
  bars [1, 2, 3, 4, 5, 6], sum(deposit.amount month of addmonths(today, it - 6))
form member, "Add member"
list deposit by day
---

@base id=quran class=long_base
request: a ramadan planner for the family: daily fasts kept, tarawih, pages of quran read per person, and sehri and iftar notes
---
app "Ramadan" moon plum
table person name text
table day_log person ref person, fasted bool, tarawih bool, pages num 0..60 = 0, note text?
setting target_pages num 1..60 = 20 "Pages a day goal"
state who ref person
let done_pages = sum(day_log.pages today)
topbar person, "Add person"
choice who
"Today" head
row gap 2
  statcard count(day_log today where fasted = yes), "fasting today"
  statcard count(day_log today where tarawih = yes), "prayed tarawih"
goal done_pages, target_pages * count(person), "pages read by the family today", "pages"
section "Log for", "pick a person above"
  toggle day_log today where person = who and fasted = yes
  toggle day_log today where person = who and tarawih = yes
  row gap 2
    "Pages today" grow
    button "+1" tap add day_log person = who, pages = 1
    button "+5" tap add day_log person = who, pages = 5
section "Pages per person", "this month"
  each person sort name as p
    row gap 2
      p.name grow
      sum(p.day_log.pages month) bold
section "Fasts kept", "this month"
  each person sort name as p
    row gap 2
      p.name grow
      count(p.day_log month where fasted = yes) "of 30" faint
section "Family pages per day"
  daybars last 14 days, sum(day_log.pages on it)
section "Notes"
  each day_log where note != "" as d
    row card gap 2 swipe remove d
      d.person.name bold
      d.note grow
form day_log, "Add a note"
---
