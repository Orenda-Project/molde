needs: monthly spot rent | cost per cup | selling price per cup | cups per day to cover rent | profit at planned cups
---
app "Chai Cart" cup orange
state rent money = 15000
state costc money = 18
state sellc money = 50
state plan num = 150
let need = if(sellc > costc, rent / 30 / (sellc - costc), 0)

    rect 0, 0, 100 * clamp(plan / max(need * 2, 1), 0, 1), 6 fill if(plan >= need, accent, warn) radius 3
    line 50, 0, 50, 6 stroke ink thick 1
  "Line marks the break-even point." faint
  "Daily profit after rent:" round(plan * (sellc - costc) - rent / 30) bold color if(plan >= need, good, bad)

  "Selling price must be above the cost per cup." color bad
