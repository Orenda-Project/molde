app "Renewals" tag yellow
table vendor name text, cost money, cycle pick Monthly|Yearly = Yearly, renews date
let yearly(v vendor) = if(v.cycle = Monthly, v.cost * 12, v.cost)
topbar "Vendor renewals", vendor, "Add vendor"
remind "A licence renews in two weeks" at vendor.renews - 14d at 9:00
statcard sum(vendor, yearly(it)), "paid to vendors per year"
section "Biggest first", "per year"
  bars vendor sort yearly(it) desc top 6, yearly(it), it.name
each vendor sort renews as v empty "No vendors yet. Add every tool and licence you pay for."
