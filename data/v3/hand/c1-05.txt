@rich domain=c1_two_bank_loans topic=compare_two_bank_loan_offers style=detailed
two banks offered me a car loan. same amount and years but different yearly rates and different processing fees in percent. show the monthly installment of each, the total i will pay including the fee, and tell me which bank is cheaper
needs: loan amount and years | rate of each bank | processing fee percent of each bank | monthly installment of each | total cost and cheaper bank
---
app "Bank Compare" money slate
state loan money = 0
state years num = 5
state ra num = 15
state fa num = 1
state rb num = 13.5
state fb num = 2.5
let n = years * 12
let emi(r num) = if(r = 0, loan / n, loan * r / 1200 * pow(1 + r / 1200, n) / (pow(1 + r / 1200, n) - 1))
let ta = emi(ra) * n + loan * fa / 100
let tb = emi(rb) * n + loan * fb / 100
row gap 2
  field loan "Loan"
  field years "Years"
section "Bank A"
  row gap 2
    field ra "Rate %"
    field fa "Fee %"
section "Bank B"
  row gap 2
    field rb "Rate %"
    field fb "Fee %"
if loan > 0
  statcard round(abs(ta - tb)), if(ta <= tb, "cheaper at Bank A", "cheaper at Bank B")
  row gap 2
    stat round(emi(ra)), "Bank A per month"
    stat round(emi(rb)), "Bank B per month"
else
  "Type the loan amount to compare the banks." faint
===
