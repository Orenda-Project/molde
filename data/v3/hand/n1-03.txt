table cost what text, amount money
state d date = today
let s = sum(sale.amount on d)
let c = sum(cost.amount on d)
topbar "Day Close", sale, "Add sale"
row align center gap 1
  button "<" tap set d = d - 1
  text d bold grow align center
  button ">" tap set d = d + 1
text s - c huge
if(s >= c, "In profit on this day", "A loss on this day") faint color if(s >= c, good, bad)
draw 100, 16
  rect 0, 1, 100 * s / max(max(s, c), 1), 6 fill accent radius 3
  rect 0, 9, 100 * c / max(max(s, c), 1), 6 fill warn radius 3
"Sales" s ", expenses" c faint
"Seven day average sales:" round(sum(sale.amount last 7 days) / 7) faint
form cost, "Add an expense"
list sale on d, "No sales on this day."
