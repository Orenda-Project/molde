let slots(b beast) = if(b.kind = Goat, 1, 7)
screen beasts "Animals" paw
  topbar beast, "Add animal"
  statcard count(beast, count(it.part) < slots(it)), "animals not full yet"
  each beast sort tag as b empty "No animals yet. Add the first one."
    row card gap 2 tap edit b swipe remove b
      stack grow gap 0
        b.kind b.tag bold
        "Share costs" round(b.price / slots(b)) faint
      slots(b) - count(b.part) "left" bold color if(slots(b) = count(b.part), good, warn)
screen shares "Shares" users
  topbar part, "Add share"
  each part sort at desc as s empty "No shares yet."
    row card gap 2 tap edit s swipe remove s
      s.who bold grow
      s.beast.tag faint
