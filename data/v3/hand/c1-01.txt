@rich domain=c1_discount_ladder topic=stacked_discounts style=custom_ui
sale season, stores stack discounts like 20% off then extra 10% off then 5% coupon. let me type the price and set each step, show the price after every step and the real total discount
needs: type the original price | three discount steps | price after each step | final price | real total discount percent
---
app "Discount Ladder" tag pink
state price money = 0
state d1 num = 20
state d2 num = 10
state d3 num = 5
let p1 = price * (1 - d1 / 100)
let p2 = p1 * (1 - d2 / 100)
let p3 = p2 * (1 - d3 / 100)
field price "Original price"
"First discount" d1 "%" faint
slider d1 0..80
"Extra discount" d2 "%" faint
slider d2 0..50
"Coupon" d3 "%" faint
slider d3 0..30
if price > 0
  section "Step by step", "total off" round((1 - p3 / price) * 1000) / 10 "%"
    row
      "After first" grow
      round(p1 * 100) / 100
    row
      "After extra" grow
      round(p2 * 100) / 100
    row
      "After coupon" grow
      round(p3 * 100) / 100 big
else
  "Type the price to climb the ladder." faint
