app "Bee Practice" star mint
table word spelling text
table try word ref word, ok bool
topbar "Bee Practice", word, "Add word"
let hits = count(try today where ok = yes)
statcard round(hits * 100 / max(count(try today), 1)), "% right today", 100
each word sort count(it.try where ok = no) - count(it.try where ok = yes) desc top 1 as w empty "No words yet. Add the list from your coach."
  stack card gap 2 align center
    "Say it, then spell it" faint
    w.spelling huge
    row gap 1
      button "Missed" fill bad tap add try word = w, ok = no
      button "Got it" fill good tap add try word = w, ok = yes
section "All words"
  each word sort count(it.try where ok = no) desc as w
    row gap 2 tap edit w swipe remove w
      w.spelling grow
      count(w.try where ok = yes) "right" faint color good
      count(w.try where ok = no) "missed" faint color bad
