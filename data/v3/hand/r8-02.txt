needs: add a booking with group, date, people, tour, price and deposit | income expected this month | balance still owed by groups | upcoming bookings sorted by date | days until each booking
---
app "Tour Bookings" map teal
table trip group text, going date, people num 1..60 = 4, tour pick Attabad|Baltit|Khunjerab = Attabad, price money, deposit money = 0
let total = sum(trip, it.people * it.price)
topbar "Bookings", trip, "Add booking"
row gap 2
  statcard sum(trip month, it.people * it.price), "income this month"
  statcard total - sum(trip.deposit), "still owed by groups"
each trip where going >= today sort going as t empty "No upcoming tours. Add the next booking."
  row card gap 2 tap edit t swipe remove t
    stack grow gap 0
      t.group bold
      t.tour "·" plural(t.people, "person") faint
    stack gap 0 align end
      t.going bold
      plural(t.going - today, "day") faint
===

van hire bookings with fare and advance, show balance due
needs: add a booking with vehicle, date, route and fare | advance received | balance due per booking | total balance due
---
app "Van Hire" bus slate
table hire route text, vehicle pick Van|Coaster|Bus = Van, going date, fare money, advance money = 0
topbar "Bookings", hire, "Add booking"
statcard sum(hire.fare) - sum(hire.advance), "balance due from customers"
each hire sort going as h empty "No bookings yet. Add the first trip."
  row card gap 2 tap edit h swipe remove h
    stack grow gap 0
      h.route bold
      h.vehicle h.going faint
    stack gap 0 align end
      h.fare - h.advance bold color if(h.fare > h.advance, bad, good)
      "advance" h.advance faint
