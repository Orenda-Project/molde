setting work num 5..600 = 30 "Work seconds"
setting rest num 5..300 = 10 "Rest seconds"
table round_done
state running bool = no
state phase pick Work|Rest = Work
state left num = 30

when running and left <= 0 and phase = Work set phase = Rest then set left = rest
when running and left <= 0 and phase = Rest set phase = Work then set left = work then add round_done

  circle 44 stroke soft thick 7
  arc 44, 0, 360 * left / if(phase = Work, work, rest) stroke if(phase = Work, accent, good) thick 7
  label 50, 44, left huge
  label 50, 66, phase small
row gap 2
  button if(running, "Pause", "Start") fill accent grow tap set running = not running
  button "Reset" grow tap set running = no then set phase = Work then set left = work

section "Last 7 days"
  daybars last 7 days, count(round_done on it)
