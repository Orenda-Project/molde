setting discount num 0..100 = 0 "Discount %"
setting tax num 0..50 = 0 "Tax %"
state a money = 0
state n num = 1
let sub = sum(line, it.qty * it.unit)
let off = sub * discount / 100
let due = (sub - off) * (1 + tax / 100)
section "Quick price", "price x quantity"
  field a "Price each"
  "Quantity" n faint
  slider n 1..50
  a * n big
topbar "Quote", line, "Add item"
each line sort at as l empty "No items yet. Add what the customer asked for."
  row card gap 2 tap edit l swipe remove l
    stack grow gap 0
      l.item bold
      l.qty "x" l.unit faint
    l.qty * l.unit bold
section "Total", "discount" discount "% · tax" tax "%"
  "Subtotal" sub
  "Discount" off
  "To charge" round(due) big
