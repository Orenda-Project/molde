needs: add a language with total strings and translated strings | buttons to add 10 translated strings | bar per language sorted by percent | overall completion | list of languages fully done
---
app "Translations" map plum
table lang name text, total num 1.., done num 0.. = 0 "Translated"
let pct(l lang) = round(l.done * 100 / max(l.total, 1))
topbar "Translations", lang, "Add language"
statcard round(sum(lang.done) * 100 / max(sum(lang.total), 1)), "% of all strings translated", 100
each lang sort round(it.done * 100 / max(it.total, 1)) desc as l empty "No languages yet. Add the first one with its string count."
  stack card gap 1 swipe remove l tap edit l
    row gap 1
      l.name bold grow
      pct(l) "%" bold
    draw 100, 5
      rect 0, 0, 100, 5 fill soft radius 2
      rect 0, 0, 100 * clamp(l.done / l.total, 0, 1), 5 fill if(pct(l) >= 100, good, accent) radius 2
    button "+10 strings" tap set l.done = min(l.done + 10, l.total)
"Fully translated:" count(lang where done >= total) faint
===
@rich domain=d3_cve_advisories topic=security_advisory_tracker style=custom_ui
security advisories for our dependencies: CVE id, package, severity low medium high critical, the date it was reported, status open or patched. highlight open ones older than 30 days, filter by severity and show open counts per severity
needs: add an advisory with CVE id, package, severity and date | status open or patched | highlight open ones older than 30 days | filter by severity | open count per severity
---
app "Advisories" shield red dark
table cve code text "CVE", pkg text, sev pick Low|Medium|High|Critical = Medium, reported date = today, status pick Open|Patched|Wontfix = Open
state s pick All|Low|Medium|High|Critical = All
topbar "Advisories", cve, "Add advisory"
grid 4 gap 1
  each options cve.sev as v
    stack radius 10 pad 1 gap 0 align center fill if(v = Critical, bad, soft)
      count(cve where sev = v and status = Open) big
      v tiny faint
choice s
each cve where status = Open and (s = All or sev = s) sort reported as c empty "No open advisories here. Keep dependencies updated."
  row card gap 2 tap edit c swipe remove c fill if(today - c.reported > 30, warn, soft)
    stack grow gap 0
      c.code c.pkg bold
      c.sev "· open" plural(today - c.reported, "day") faint
    button "Patched" tap set c.status = Patched
