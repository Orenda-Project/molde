app "Grandparents" phone pink
table contact who pick Nani|Nana|Dadi|Dada = Nani, kind pick Call|Video|Visit = Call, mins num 1..300 = 10
remind "Have you called a grandparent today?" at 18:00 if count(contact today) = 0
topbar "Grandparents", contact, "Log contact"
statcard sum(contact.mins week), "minutes with grandparents this week"
each options contact.who as w
  stack card gap 1
    row gap 2 align center
      text w head grow
      if count(contact where who = w) = 0
        "not yet" faint
      else
        since(contact where who = w) "days ago" bold color if(due(contact where who = w, 7), bad, good)
    row gap 1
      each options contact.kind as k
        button k grow tap add contact who = w, kind = k
