app "Parcels" package mint
table parcel what text, who text, sent date = today, arrived bool
topbar parcel, "Add parcel"
statcard count(parcel where arrived = no), "parcels on the way"
each parcel sort arrived as p empty "No parcels yet. Add the one you just posted."
  row card gap 2 tap edit p swipe remove p
    toggle p.arrived
    stack grow gap 0
      p.what bold
      "For" p.who faint
    if p.arrived
      "Arrived" faint color good
    else
      plural(today - p.sent, "day") bold color if(today - p.sent > 7, bad, ink)
