table line place text, now_text text "Current", better text "Proposed", status pick Open|Agreed|Shipped = Open
state q text = ""
topbar "Microcopy review", line, "Add text"
count(line where status = Open) "texts waiting for a decision" head
field q "Search texts"
each line where has(place, q) or has(now_text, q) or has(better, q) as l empty "No texts here. Add a button label or an error message to review."
  stack card gap 1 tap edit l swipe remove l
    l.place faint
    l.now_text "→" l.better bold
    row gap 1
      l.status faint grow
      if l.status = Open
        button "Agree" tap set l.status = Agreed

app "Icon queue" star red
table icon name text, size pick S|M|L = M, who text?, due date, sent bool
let late = count(icon where sent = no and due < today)
topbar "Icon requests", icon, "Add request"
statcard count(icon where sent = no), "icons to draw," late "overdue"
remind "An icon is due tomorrow" at icon.due - 1d at 9:00
each icon where sent = no sort due as i empty "No icons waiting. Anything new from the product team goes here."
  row card gap 2 tap edit i swipe remove i fill if(i.due < today, bad, soft)
    toggle i.sent
    stack grow gap 0
      i.name bold
      i.size "· for" i.who "· due" i.due faint
