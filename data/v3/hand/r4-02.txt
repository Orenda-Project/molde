table venue name text, abstract date, paper date, status pick Idea|Writing|Submitted|Accepted|Rejected = Idea
remind "Paper deadline in a week" at venue.paper - 7d at 9:00

  stack grow gap 0
    nxt.name bold
    "next paper deadline" faint

    stat count(venue where status = it), it

    row
      v.name bold grow
