app "Gym Value" dumbbell blue
table visit
setting fee money = 6000 "Monthly fee"
setting target num = 12 "Visits wanted"
let n = count(visit month)
button "I went today" fill accent tap add visit
goal n, target, "visits this month"
row gap 2
  statcard n, "visits so far"
  statcard if(n > 0, round(fee / n), fee), "per visit so far"
if n >= target
  "Target reached. Every extra visit now makes it cheaper." bold color good
else
  "Still" target - n "visits to go." faint
daygrid last 28 days, count(visit on it), 1
