app "Body tape" scale pink
table tape waist num 30..200 unit "cm", chest num 30..200 unit "cm", hips num 30..200 unit "cm", arm num 15..80 unit "cm"
state part pick Waist|Chest|Hips|Arm = Waist
let v(r tape) = if(part = Waist, r.waist, if(part = Chest, r.chest, if(part = Hips, r.hips, r.arm)))
topbar "Body tape", tape, "Measure"
choice part
row gap 2
  statcard v(last(tape)), "cm now"
  statcard round((v(last(tape)) - v(first(tape))) * 10) / 10, "cm change since the first"
section part, "every measurement"
  trend tape sort at, v(it)
each tape sort at desc top 6 as t empty "No measurements yet. Take the first tape reading."
  row card gap 2 tap edit t swipe remove t
    t.at grow
    "W" t.waist "· C" t.chest "· H" t.hips "· A" t.arm faint
