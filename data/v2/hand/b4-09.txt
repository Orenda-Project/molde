@create style=detailed domain=b4_menu_costing
restaurant dishes with their ingredient costs so I know my margin. each dish has a selling price, and ingredients lines with the cost for one plate. show each dish with cost, price and margin percent, worst margin first
---
app "Menu Costing" food slate
table dishc name text, price money
table ingr dishc ref dishc, item text, cost money
each dishc sort round(100 * (it.price - sum(it.ingr.cost)) / it.price) as d
  let cost = sum(d.ingr.cost)
  stack card gap 1 tap edit d
    row
      d.name bold grow
      text round(100 * (d.price - cost) / d.price) "% margin" color if(d.price - cost < d.price * 0.3, warn, good)
    text "Cost" cost "Price" d.price faint
form ingr, "Add ingredient"
form dishc, "Add dish"
===

+ bars options dishc.course, avg(dishc.price where course = it), it
===
