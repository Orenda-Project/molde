@create style=detailed domain=b2_running_shoes
I run 5 days a week and I keep killing shoes. Log each run with km and which pair I wore, add up the km on each pair, and warn me when a pair passes 600 km.
---
app "Run log" run mint
table shoe brand text, retire_at num 100..1500 = 600 "Retire at (km)"
table run shoe ref shoe, km num 0.5..100, minutes num 1..600
sum(run.km week) "km this week" big
each shoe as s
  stack card gap 1 tap edit s
    row
      s.brand bold grow
      sum(s.run.km) "km" bold color if(sum(s.run.km) >= s.retire_at, bad, ink)
    if sum(s.run.km) >= s.retire_at
      "Time for new shoes" color bad small
    "Average pace" round(sum(s.run.minutes) / max(sum(s.run.km), 1)) "min per km" faint
form shoe, "Add shoes"
form run, "Log run"
===
@create style=custom_ui domain=b2_kabaddi_board

---

    "Home" faint

    button "Raid +1" tap set away = away + 1
    button "Tackle +2" tap set away = away + 2
