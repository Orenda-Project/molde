@create style=detailed domain=b2_homeschool_hours

---

  stack card gap 1

@create style=typos domain=b2_family_bmi
bmi for everyone in the family, hight and weight, and tell if underweight normal or over
---
app "Family BMI" scale mint
table member name text, cm num 50..220, kg num 3..200
each member sort name as m
  let bmi = m.kg / pow(m.cm / 100, 2)
  row card tap edit m
    m.name bold grow
    text fixed(bmi, 1) big
    if(bmi < 18.5, "Under", if(bmi < 25, "Normal", "Over")) color if(bmi < 25 and bmi >= 18.5, good, warn)
