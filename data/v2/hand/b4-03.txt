@create style=detailed domain=b4_butcher_counter
halal meat shop: beef, mutton and chicken have different rates per kg that change every week. weigh and sell, show money per meat today, and a second tab for the rates and the week summary
---
app "Meat Shop" food slate
table sale meat pick Beef|Mutton|Chicken = Beef, kg num 0.1..50, amount money
setting beef_rate money = 1100 "Beef per kg"
setting mutton_rate money = 2200 "Mutton per kg"
setting chicken_rate money = 550 "Chicken per kg"
state meat_pick pick Beef|Mutton|Chicken = Beef
state weight num = 1
let rate = if(meat_pick = Beef, beef_rate, if(meat_pick = Mutton, mutton_rate, chicken_rate))
screen counter "Counter" food
  choice meat_pick
  field weight "Weight (kg)"
  text "Price" round(rate * weight) big
  button "Sell" tap add sale meat = meat_pick, kg = weight, amount = rate * weight
  each options sale.meat as m
    row card
      m bold grow
      sum(sale.kg today where meat = m) "kg"
      sum(sale.amount today where meat = m) bold
screen week "Week" chart
  "This week" sum(sale.amount week) big
  daybars last 7 days, sum(sale.amount on it)
  field beef_rate
  field mutton_rate
  field chicken_rate
===

@create style=multi_feature domain=b4_biryani_pots
