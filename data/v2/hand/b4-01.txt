@create style=detailed domain=b4_restaurant_tables
we run a small dhaba with about 12 tables. waiters punch in what each table ordered, and at the end i need the bill for one table at a time, only the unpaid items, then a button to close the table
---
app "Table Bills" food ocean
table dish name text, price money
table order table_no num 1..30, dish ref dish, qty num 1..20 = 1, status pick Open|Paid = Open
state open_table num = 1
let bill = sum(order where table_no = open_table and status = Open, it.qty * it.dish.price)
row gap 1
  button "<" tap set open_table = max(open_table - 1, 1)
  text "Table" open_table head grow align center
  button ">" tap set open_table = min(open_table + 1, 30)
"Bill" bill big
each order where table_no = open_table and status = Open as o
  row card tap edit o swipe remove o
    o.dish.name grow
    o.qty "x" o.dish.price
button "Close table" tap set all order.status = Paid where table_no = open_table
form dish, "Add dish to menu"
form order, "Add order"
===
