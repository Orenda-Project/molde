@create style=custom_ui domain=b3_emi_calc
car loan calculator, price, down payment percent, years with a slider and yearly interest, shows monthly instalment, total paid and total interest
---
app "Loan Calc" calc slate
state price num = 3000000
state down num = 20
state years num = 5
state yearly_rate num = 18
let loan = price * (100 - down) / 100
let r = yearly_rate / 1200
let n = years * 12
let emi = loan * r * pow(1 + r, n) / (pow(1 + r, n) - 1)
field price "Price"
field down "Down payment %"
slider years 1..10
field yearly_rate "Yearly rate %"
"Monthly instalment" round(emi) big
"Total paid" round(emi * n + price * down / 100) faint
"Interest paid" round(emi * n - loan) faint
===
