@create style=custom_ui domain=b3_basketball
basketball scoreboard for home and away, buttons for 1 2 3 points, pick the quarter with chips, big totals, undo
---
app "Basketball" ball ink
table bpoint team pick Home|Away = Home, pts num 1..3, quarter num 1..4
state period num = 1
let home_total = sum(bpoint.pts where team = Home)
let away_total = sum(bpoint.pts where team = Away)
row gap 1
  each [1, 2, 3, 4] as q
    button q fill if(period = q, accent, soft) tap set period = q
row gap 2 align center
  stack grow align center
    "Home" faint
    home_total huge
  stack grow align center
    "Away" faint
    away_total huge
row gap 1
  each [1, 2, 3] as n
    button "Home +" n tap add bpoint team = Home, pts = n, quarter = period
row gap 1
  each [1, 2, 3] as n
    button "Away +" n tap add bpoint team = Away, pts = n, quarter = period
"This quarter" sum(bpoint.pts where quarter = period) faint
button "Undo" tap remove last(bpoint)
===

@create style=detailed domain=b3_dominoes
