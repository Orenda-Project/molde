@create style=multi_feature domain=b1_feed_dealer
poultry feed dealer. I sell bags to farms on credit, each farm has a ledger. I record each bag sale with a price and payments received, and want the balance per farm with the farm with highest balance on top
---
app "Feed Dealer" package ocean
table farm name text, owner text?
table bagsale farm ref farm, bags num 1..500, rate money
table receipt farm ref farm, amount money, mode pick Cash|Bank|Cheque = Cash
each farm sort sum(it.bagsale, it.bags * it.rate) - sum(it.receipt.amount) desc as f
  let balance = sum(f.bagsale, it.bags * it.rate) - sum(f.receipt.amount)
  row card tap edit f
    stack grow
      f.name bold
      f.owner faint
    balance bold color if(balance > 0, bad, good)
"Market outstanding" sum(bagsale, it.bags * it.rate) - sum(receipt.amount) head
form bagsale, "Bag sale"
form receipt, "Payment"
form farm, "Add farm"
