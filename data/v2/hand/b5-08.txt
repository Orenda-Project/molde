---

table roll name text, metres num 1..500
table cutlen roll ref roll, m num 0.25..50
each roll sort name as r
  let left = r.metres - sum(r.cutlen.m)
  stack card gap 1 tap edit r

      r.name bold grow
      fixed(left, 1) "m left" bold color if(left < 3, bad, ink)
