@create style=custom_ui domain=b3_darts
darts 501: each player's card shows what's left, quick buttons for the usual scores 26 41 45 60 100 and a box for any other score
---
app "Darts 501" target ink
table player name text
table throw player ref player, score num 0..180
each player as p
  let left = 501 - sum(p.throw.score)
  stack card gap 1
    row
      p.name head grow
      left big color if(left = 0, good, ink)
    "Visits" count(p.throw) "| average" fixed(sum(p.throw.score) / count(p.throw), 1) faint
    row gap 1
      each [26, 41, 45, 60, 100] as sc
        button sc tap add throw player = p, score = sc
form player, "Add player"
form throw, "Add exact score"
===
