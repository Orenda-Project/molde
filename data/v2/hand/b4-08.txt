@create style=short domain=b4_breakfast_stall
halwa puri breakfast stall, plates sold and ingredient cost every day, so I see profit
---
app "Breakfast Stall" food mint
table plates qty num 1..200 = 1
table cost_b item pick Flour|Oil|Chana|Semolina|Gas|Other = Flour, amount money
setting plate_price money = 180 "Price per plate"
let income = sum(plates.qty today) * plate_price
let spent = sum(cost_b.amount today)
text "Profit today" income - spent big color if(income >= spent, good, bad)
row gap 2
  stat sum(plates.qty today), "plates"
  stat spent, "spent"
button "Plate sold" tap add plates qty = 1
form cost_b, "Bought"
