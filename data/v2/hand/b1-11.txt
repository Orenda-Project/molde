@create style=detailed domain=b1_cotton_pickers
cotton season: pickers pick kg for me and I pay per kg at a rate that I can change. record kg picked by each picker, show their pay this week and total kg in the season
---
app "Cotton Pickers" leaf mint
table picker name text
table picked picker ref picker, kg num 1..200
table picker_paid picker ref picker, amount money
setting pick_rate money = 40 "Pay per kg"
each picker sort name as p
  let owed = sum(p.picked.kg) * pick_rate - sum(p.picker_paid.amount)
  row card tap edit p
    p.name bold grow
    text sum(p.picked.kg week) "kg this week"
    owed bold color if(owed > 0, warn, good)
"Season total" sum(picked.kg) "kg" big
form picked, "Kg picked"
form picker_paid, "Paid"
form picker, "Add picker"
===
