@create style=detailed domain=b4_food_truck_events
my food truck goes to different events, each event charges a stall fee, and I note sales amount per event. show profit per event after the fee and the best event so far
---
app "Food Truck" car slate
table fevent name text, day date, fee money
table fsale fevent ref fevent, amount money
each fevent sort day desc as e
  let profit = sum(e.fsale.amount) - e.fee
  row card tap edit e
    stack grow
      e.name bold
      text e.day faint
    text profit bold color if(profit >= 0, good, bad)
text "Best event" max(fevent, sum(it.fsale.amount) - it.fee) big
form fsale, "Sale at event"
form fevent, "New event"
