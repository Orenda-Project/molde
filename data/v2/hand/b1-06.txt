@create style=detailed domain=b1_broiler_cycle
broiler cycle accounting: I put chicks, feed, medicine and labour costs against a flock, and then record sales of birds with kg and rate. show cost, income and profit for each flock
---
app "Broiler Cycle" egg mint
table flock name text, birds num 100..20000
table cost flock ref flock, kind pick Chicks|Feed|Medicine|Labour|Electricity = Feed, amount money
table birdsale flock ref flock, kg num 1..5000, rate money
each flock sort at desc as f
  let spent = sum(f.cost.amount)
  let earned = sum(f.birdsale, it.kg * it.rate)
  stack card gap 1 tap edit f
    row
      f.name bold grow
      text f.birds "birds" faint
    row gap 2
      text "Cost" spent
      text "Income" earned
      text "Profit" earned - spent bold color if(earned >= spent, good, bad)
    "Cost per bird" round(spent / f.birds) faint
form cost, "Add cost"
form birdsale, "Birds sold"
form flock, "New flock"
===

---
