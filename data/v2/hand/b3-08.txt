@create style=custom_ui domain=b3_snakes_ladders
snakes and ladders tracker, each player has a roll button that moves them by a random dice number, show their square out of 100 with a bar
---
app "Snakes and Ladders" dice plum
table player name text
table move player ref player, steps num 1..6
each player sort name as p
  let pos = min(sum(p.move.steps), 100)
  stack card gap 1
    row
      p.name head grow
      pos "of 100" big color if(pos = 100, good, ink)
    goal pos, 100, "to the top"
    button "Roll" tap add move player = p, steps = random(6)
form player, "Add player"
===
