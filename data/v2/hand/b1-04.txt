@create style=custom_ui domain=b1_landlord_upkeep

---

table house name text
table income house ref house, amount money
table upkeep house ref house, amount money, kind pick Repair|Tax|Maintenance = Repair
each house sort name as h
  let inc = sum(h.income.amount year)
  let exp = sum(h.upkeep.amount year)
  stack card gap 1 tap edit h
    row
      h.name bold grow
      text "Net" inc - exp bold color if(inc >= exp, good, bad)
    draw 100, 8
      rect 0, 0, 100 * inc / max(max(inc, exp), 1), 3 fill good radius 1
      rect 0, 5, 100 * exp / max(max(inc, exp), 1), 3 fill bad radius 1
    text "Income" inc "| Expenses" exp faint
form income, "Rent received"
form upkeep, "Expense"
form house, "Add house"
