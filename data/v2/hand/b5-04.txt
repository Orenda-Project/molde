@create style=detailed domain=b5_date_palms
date palm orchard: every palm has a tag number and variety (Ajwa, Khudri, Mabroom or Local). at harvest I weigh dates per palm. show total kg per variety, and which palm gave the most
---
app "Palm Grove" tree mint
table palm tag text, variety pick Ajwa|Khudri|Mabroom|Local = Local
table cutting palm ref palm, kg num 0.5..400
"Total harvest" sum(cutting.kg year) "kg" big
each options palm.variety as v
  row
    v bold grow
    sum(cutting year, if(it.palm.variety = v, it.kg, 0)) "kg"
"Top palm" head
each palm sort sum(it.cutting.kg) desc top 1 as p
  row card
    p.tag bold grow
    p.variety faint
    sum(p.cutting.kg) "kg" bold
form cutting, "Weigh harvest"
form palm, "Add palm"
===

@create style=vague domain=b5_juice_margin

@create style=custom_ui domain=b5_qurbani_shares

---

  stack card gap 1
    row
      a.name bold grow
      a.price / 7 "per share" faint

        rect (n - 1) * 14.3, 0, 12, 12 fill if(n <= taken, accent, soft) radius 2
