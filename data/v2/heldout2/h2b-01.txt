@create style=detailed domain=h2b_loan_emi
loan EMI calculator: loan amount, yearly interest percent and number of months, shows the monthly payment, total I will pay and total interest
---
app "EMI" calc slate
setting amount num 1000.. = 500000 "Loan amount"
setting rate num 0..60 = 12 "Interest per year (%)"
setting months num 1..360 = 24 "Months"
field amount
field rate
field months
let r = rate / 1200
let emi = if(r = 0, amount / months, amount * r * pow(1 + r, months) / (pow(1 + r, months) - 1))
stat round(emi), "per month"
"Total" round(emi * months) "Interest" round(emi * months - amount)
