@create style=multi_feature domain=h2b_flashcards
flashcards for my driving test. front question, back answer. show one card at a time, tap to reveal the answer, and a next button
---
app "Driving cards" cards ink
table card front text, back text
state i num = 1
state show bool = no
let n = count(card)
let q = nth(card sort front, i).front
let a = nth(card sort front, i).back
if n > 0
  stack card gap 1
    text q big
    if show
      text a
    else
      "Tap Show answer" faint
  row gap 1
    button if(show, "Hide", "Show answer") tap set show = not show
    button "Next" tap set show = no then set i = i % n + 1
form card, "Add card"
===
