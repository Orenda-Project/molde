@create style=custom_ui domain=h2a_flash
flashcards for english words, show the word, tap to reveal meaning, next button. random order
---
app "Flashcards" cards ocean
table card front text, back text
state i num = 1
state shown bool = no
let cur = nth(card shuffle, i)
form card
if count(card, yes) > 0
  stack card gap 1 tap set shown = not shown
    text cur.front head
    if shown
      text cur.back
  button "Next" tap set i = i % count(card, yes) + 1 then set shown = no
===

@create style=multi_feature domain=h2a_udhar

---

table person name text
table entry person ref person, amount money, kind pick Gave|Got = Gave
form person
form entry
each person sort sum(it.entry.amount where kind = Gave) - sum(it.entry.amount where kind = Got) desc
  row card
    it.name bold grow
